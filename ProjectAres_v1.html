    let currentIndex = -1;

    let score = 0;


    function loadScenario() {

        if (currentIndex >= database.length) { endGame(); return; }


        const currentData = database[currentIndex];

        

        document.getElementById("hud-progress").innerText = (currentIndex + 1) + " / " + database.length;

        document.getElementById("domain").innerText = currentData.domain + " - " + currentData.topic;

        document.getElementById("speaker").innerText = currentData.speaker;

        document.getElementById("char").innerText = currentData.char;

        document.getElementById("message").innerText = currentData.scenario;

        

        document.getElementById("next-btn").style.display = "none";

        const optionsBox = document.getElementById("options");

        optionsBox.style.display = "flex";

        optionsBox.innerHTML = ""; 


        let choices = [

            { text: currentData.correct, isCorrect: true },

            { text: currentData.wrong1, isCorrect: false },

            { text: currentData.wrong2, isCorrect: false }

        ];

        choices.sort(() => Math.random() - 0.5); // Shuffle


        choices.forEach(choice => {

            let btn = document.createElement("button");

            btn.className = "option-btn";

            btn.innerText = choice.text;

            btn.onclick = () => makeChoice(choice.isCorrect, currentData.feedback);

            optionsBox.appendChild(btn);

        });

    }


    function makeChoice(isCorrect, feedback) {

        document.getElementById("options").style.display = "none";

        if (isCorrect) {

            score += 100;

            document.getElementById("message").innerHTML = "✅ <strong>CORRECT!</strong><br><br>" + feedback;

            document.getElementById("char").innerText = "🎉";

        } else {

            document.getElementById("message").innerHTML = "❌ <strong>INCORRECT!</strong><br><br>" + feedback;

            document.getElementById("char").innerText = "🤦";

        }

        document.getElementById("hud-score").innerText = score;

        document.getElementById("speaker").innerText = "System Feedback";

        document.getElementById("next-btn").style.display = "block";

    }


    function nextQuestion() {

        currentIndex++;

        loadScenario();

    }
